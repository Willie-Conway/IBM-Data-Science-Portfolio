# Comprehensive data cleaning pipeline def clean_spacex_data(df): """Complete data cleaning and preparation pipeline""" # 1. Standardize categorical variables df['Orbit'] = df['Orbit'].str.upper().str.strip() df['LaunchSite'] = df['LaunchSite'].str.upper().str.strip() # 3. Create derived features df['Year'] = pd.to_datetime(df['Date']).dt.year df['Month'] = pd.to_datetime(df['Date']).dt.month df['DaysSinceFirstLaunch'] = (pd.to_datetime(df['Date']) - pd.to_datetime(df['Date']).min()).dt.days # 4. Validate data types numerical_columns = ['FlightNumber', 'PayloadMass', 'Flights', 'Block', 'ReusedCount'] for col in numerical_columns: df[col] = pd.to_numeric(df[col], errors='coerce') return df

⚙️ Feature Engineering for ML

Created 80+ engineered features from raw launch data to enhance predictive modeling capabilities.

# Feature engineering pipeline for machine learning def engineer_features(df): """Create predictive features from launch data""" features = {} # 1. Temporal Features features['Year'] = df['Date'].dt.year features['Month'] = df['Date'].dt.month features['Quarter'] = df['Date'].dt.quarter features['DaysSinceFirstLaunch'] = (df['Date'] - df['Date'].min()).dt.days # 2. Experience Features features['FlightNumber'] = df['FlightNumber'] features['CumulativeLaunches'] = df['FlightNumber'].cumsum() features['SuccessStreak'] = df['Class'].rolling(window=5, min_periods=1).sum() # 3. Payload Features features['PayloadMass'] = df['PayloadMass'] features['PayloadToOrbitRatio'] = df['PayloadMass'] / df.groupby('Orbit')['PayloadMass'].transform('mean') # 4. Booster Features features['BoosterVersionNum'] = df['BoosterVersion'].str.extract(r'(\d+)').astype(float) features['IsReused'] = df['Reused'].astype(int) features['ReuseCount'] = df['ReusedCount'] # 5. Site Features (One-hot encoded) site_dummies = pd.get_dummies(df['LaunchSite'], prefix='Site') features.update(site_dummies.to_dict('series')) # 6. Orbit Features (One-hot encoded) orbit_dummies = pd.get_dummies(df['Orbit'], prefix='Orbit') features.update(orbit_dummies.to_dict('series')) return pd.DataFrame(features)

End-to-end machine learning pipeline for landing success prediction.

# Complete ML Pipeline Implementation def landing_success_pipeline(df): """End-to-end landing success prediction pipeline""" # 1. Feature Engineering features = engineer_features(df) # Creates 83 features target = df['Class'] # 2. Data Preparation scaler = StandardScaler() X_scaled = scaler.fit_transform(features) # 3. Train-Test Split X_train, X_test, y_train, y_test = train_test_split( X_scaled, target, test_size=0.2, random_state=42 ) # 4. Model Training with GridSearchCV models = { 'decision_tree': GridSearchCV( DecisionTreeClassifier(), param_grid={ 'max_depth': [5, 10, 12, 15], 'criterion': ['gini', 'entropy'] }, cv=10 ), 'svm': GridSearchCV( SVC(), param_grid={ 'C': [0.01, 0.1, 1.0, 10], 'kernel': ['linear', 'rbf', 'sigmoid'] }, cv=10 ) } # 5. Model Evaluation results = {} for name, model in models.items(): model.fit(X_train, y_train) train_score = model.score(X_train, y_train) test_score = model.score(X_test, y_test) cv_score = model.best_score_ results[name] = { 'train_accuracy': train_score, 'test_accuracy': test_score, 'cv_accuracy': cv_score, 'best_params': model.best_params_ } return results
